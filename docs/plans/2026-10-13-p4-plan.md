## 0. 一页结论

```
            每 5 分钟（OpenClaw cron，command 载荷，不调模型）          语音进程启动时（后台线程）
                         │                                                   │
                         ▼                                                   ▼
             python -m guanjia.sleep check  ◄────── 同一把运行锁 ──────►  quick_reset()
                         │
  S0 读水位线(SQLite 只读) ── 空闲<60min / 忙 / 内容不够 / 冷却 ──► 退出（约 1 秒）
                         │
  S1 导出本段（检查点之后）→ S2 直连模型整理（经脱敏代理，无工具、无会话）
                         │
  S3 代码校验：只信原话（证据必须是主人/家人原句子串）· 长度上限 · 删除比例 · schema
                         │                           ┌── stage=rehearse：只写 data/sleep/staging，停
  S4 提交（wsfiles 锁 + base_sha，原子替换，留历史）──┤
                         │                           └── stage=commit：提交后停，不换新
  S5 复核水位线 + 托盘「换新中」栅栏（语音新句等 ≤10s）
                         │
  S6 sessions.reset(expectedSessionId) → 补差（毫秒窗口落进来的话）→ 写 resets.jsonl（给 P3 插件补投家人摘要）
```

- **参数**（全部在 `data/sleep-settings.json`，面板可改，下一次检查生效）：主人空闲 **60 分钟**；本段 **≥1.2 万 token 或 ≥10 轮**主人发言；两次换新 **≥2 小时**；家人支线空闲 **4 小时**（P3 启用，P4 只留接口）；强制休眠 prompt ≥6 万且空闲 ≥10 分钟（沿用 v2，第 4 节未提，默认开，面板可关）。
- **三段放量**：`stage = off → rehearse（演练 3–5 天）→ commit（提交，不换新，3 天）→ reset（真实换新）`，面板一个下拉框切换。
- **和设计 v2 的差异**：
  1. 整理请求**不走网关**，直连硅基（经本机脱敏代理 18797）：没有工具、没有系统提示开销、模型不可能顺手改文件。
  2. staging / journal / 备份放仓库 `data/sleep/`，**不放工作区**（家人支线和 memory_search 碰不到半成品）。
  3. 「只信原话」由代码强制：每条写入必须带 `evidence`，且是导出的主人/家人原话的子串，否则丢弃；出处 `src` 由代码按证据所在消息填写，模型编不了。
  4. 记忆不再是「两个大文件」：常驻核心层（USER.md + MEMORY.md 三节）+ 自动生成的目录 + 按人/主题/日期分文件的长期层；休眠负责升降（设计第 4 节）。
  5. 开机先换新不另建计划任务：语音进程每次启动都后台调一次 `quick_reset`（条件本身幂等），cron 第一次检查兜底。
- **上下文新目标**：固定开销（新会话一句话）**≤20k token**（告警 22k），其中常驻记忆约 5.9k；典型单轮 ≤45k，峰值 60k 触发强制休眠（设计第 5 节）。仍砍掉技能堆、英文默认规则、心跳、用不到的工具。

**新文件** `scripts/prompt_size.py`

```python
"""量一次 OpenClaw 的固定开销：新测试会话发一句，读 usage 和 systemPromptReport，存 tmp/prompt-size/<时间>.json。
只读线上配置；会建一个 agent:main:exp-slim-<时间> 会话，结束时删除。"""
from __future__ import annotations

import json
import sqlite3
import sys
import time
from pathlib import Path

import httpx

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "src"))
from guanjia.brain import DEFAULT_URL, load_gateway_token  # noqa: E402

try:  # P4-3 之后用包里的；P4-0 时还没有包，用本脚本里的同名实现（照抄 P4-3 oc.py 的 gateway_call 与 AGENT_DB）
    from guanjia.sleep.oc import AGENT_DB, gateway_call  # noqa: E402
except ImportError:
    from _p4_oc_fallback import AGENT_DB, gateway_call  # noqa: E402  scripts/_p4_oc_fallback.py，P4-3 合并后删除

PROBE = "[语音 2026-10-13 18:00 周二 主人] 在吗"


def report_for(key: str) -> dict:
    con = sqlite3.connect(f"file:{AGENT_DB.as_posix()}?mode=ro", uri=True, timeout=1.0)
    try:
        row = con.execute("select entry_json from session_nodes where session_key=?", (key,)).fetchone()
    finally:
        con.close()
    r = (json.loads(row[0]) if row else {}).get("systemPromptReport") or {}
    tools = (r.get("tools") or {}).get("entries") or []
    return {
        "system_chars": (r.get("systemPrompt") or {}).get("chars"),
        "project_chars": (r.get("systemPrompt") or {}).get("projectContextChars"),
        "non_project_chars": (r.get("systemPrompt") or {}).get("nonProjectContextChars"),
        "files": {f["name"]: f.get("injectedChars") for f in r.get("injectedWorkspaceFiles") or []},
        "skills": [s["name"] for s in (r.get("skills") or {}).get("entries") or []],
        "skills_chars": (r.get("skills") or {}).get("promptChars"),
        "tools": [t["name"] for t in tools],
        "tool_summary_chars": sum(t.get("summaryChars", 0) for t in tools),
        "tool_schema_chars": sum(t.get("schemaChars", 0) for t in tools),
    }


def main() -> int:
    key = f"agent:main:exp-slim-{time.strftime('%m%d%H%M%S')}"
    h = {"Authorization": f"Bearer {load_gateway_token()}", "x-openclaw-session-key": key}
    r = httpx.post(DEFAULT_URL, headers=h, json={"model": "openclaw", "messages": [{"role": "user", "content": PROBE}]},
                   timeout=120, trust_env=False)
    r.raise_for_status()
    usage = r.json().get("usage") or {}
    out = {"t": time.strftime("%Y-%m-%d %H:%M:%S"), "key": key, "usage": usage, **report_for(key)}
    try:
        gateway_call("sessions.delete", {"key": key})
    except Exception as e:  # 删不掉不影响测量结果，留着手工删
        out["delete_error"] = str(e)[:200]
    d = ROOT / "tmp" / "prompt-size"
    d.mkdir(parents=True, exist_ok=True)
    (d / f"{time.strftime('%Y%m%d-%H%M%S')}.json").write_text(json.dumps(out, ensure_ascii=False, indent=1), encoding="utf-8")
    print(json.dumps({"prompt_tokens": usage.get("prompt_tokens"), "system_chars": out["system_chars"],
                      "skills": len(out["skills"]), "tools": len(out["tools"])}, ensure_ascii=False))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
_PUNCT = re.compile(r"[\s，。！？、；：,.!?;:\"'“”‘’（）()【】\[\]…~-]+")
SECRET = re.compile(r"(?:密码|口令|sk-[A-Za-z0-9]{8,}|\b\d{15,19}\b|\b\d{17}[\dXx]\b)")
FILE_RE = re.compile(r"^(core\.user\.[^\s/]{1,20}|core\.facts|(people|topics)/[^/\\:*?\"<>|]{1,40})$")


def _norm(s: str) -> str:
    return _PUNCT.sub("", s or "")


def find_evidence(evidence: str, msgs: list[Msg]) -> Msg | None:
    """证据必须是某条人类用户行（或家人摘要）的连续子串（忽略标点空白），至少 4 个字；返回那条消息，用来填出处。"""
    ev = _norm(evidence)
    if len(ev) < 4:
        return None
    for m in msgs:
        if m.role == "user" and m.who not in ("主人?", "未知") and ev in _norm(m.text):
            return m
    return None


def src_of(m: Msg, sid: str) -> str:
    return f"d:{m.ts}" if m.who == "家人摘要" else f"s:{sid[:8]}#{m.seq}"


def filter_ops(out: dict, msgs: list[Msg], digests: list[Msg], sid: str) -> tuple[dict, list[str]]:
    """丢掉结构不对 / 证据不成立 / 含密钥 / 超长的操作，给留下的填 src；返回 (清洗后的 out, 丢弃原因)。不抛错。"""
    human = msgs + digests
    dropped: list[str] = []
    kept = []
    for op in out.get("ops") or []:
        if not isinstance(op, dict) or op.get("op") not in ("add", "rewrite", "delete") or not FILE_RE.match(str(op.get("file", ""))):
            dropped.append("ops: 结构不对"); continue
        if op["op"] in ("rewrite", "delete") and not op.get("id"):
            dropped.append("ops: 缺 id"); continue
        if op["op"] != "delete":
            text = str(op.get("text") or "")
            if not text or len(text) > caps.ITEM_MAX or SECRET.search(text):
                dropped.append(f"ops: 太长/为空/含敏感串 {text[:20]}"); continue
        m = find_evidence(str(op.get("evidence") or ""), human)
        if m is None:
            dropped.append(f"ops: 证据不是原话 {str(op.get('evidence'))[:20]}"); continue
        op["src"] = src_of(m, sid)
        op["imp"] = max(1, min(10, int(op.get("imp") or 5)))
        if op["op"] == "add" and op["imp"] <= 3:
            dropped.append(f"ops: 重要度太低 {op.get('text', '')[:20]}"); continue
        kept.append(op)
    out["ops"] = kept
    loops = []
    for lp in out.get("openLoops") or []:
        m = find_evidence(str(lp.get("evidence") or ""), human) if isinstance(lp, dict) else None
        if m is None:
            dropped.append("openLoops: 证据不是原话"); continue
        lp["src"] = src_of(m, sid)
        loops.append(lp)
    out["openLoops"] = loops
    ins = []
    for it in out.get("insights") or []:
        evs = [find_evidence(e, human) for e in (it.get("evidence") or [])] if isinstance(it, dict) else []
        hits = [m for m in evs if m is not None]
        if len({m.seq for m in hits}) < 2:
            dropped.append("insights: 少于 2 处原话"); continue
        it["src"] = ",".join(src_of(m, sid) for m in hits[:3])
        ins.append(it)
    out["insights"] = ins[:2]
    sc = out.get("soulCorpus")
    if sc and not any(m.role == "assistant" and _norm(sc.get("new", "")) and _norm(sc["new"]) in _norm(m.text) for m in msgs):
        out["soulCorpus"] = None
        dropped.append("soulCorpus: 不是助手真说过的话")
    return out, dropped
```

```python
                       core_target: Callable[[str, Item], str | None], today: date) -> list[tuple[str, str, Item]]:
    """升层候选：imp≥8 的主人长期偏好（Always/Never/Prefer 开头）→ USER 主人；14 天内 hits 增量 ≥2 → 对应核心节。
    只升放得下的（升完不超预算），按 imp、hits 降序。返回 [(长期文件, 核心节, 条目)]。"""
    out = []
    pool = []
    for rel, items_ in long_items.items():
        for it in items_:
            sec = core_target(rel, it)                 # people/主人 → core.user.主人；people/老婆 → core.user.老婆；topics/* → core.facts
            if not sec:
                continue
            pref = it.text.startswith(("Always", "Never", "Prefer")) and it.imp >= 8 and sec == "core.user.主人"
            hot = it.hits >= 2 and _days(it.lh, today) <= 14
            if pref or hot:
                pool.append((rel, sec, it))
    for rel, sec, it in sorted(pool, key=lambda x: (-x[2].imp, -x[2].hits)):
        if _size(core[sec]) + len(it.text) + 40 <= caps_by_sec[sec]:
            core[sec].append(it); out.append((rel, sec, it))
    return out
```

- `long_target(sec, it)`：`core.user.主人` → `memory/people/主人.md`；`core.user.<家人>` → `memory/people/<家人>.md`；`core.facts` → 用 `relevant.best_topic(it.text)` 找关键词最匹配的主题文件，找不到放 `memory/topics/杂项.md`。
- 升层后在长期文件里**删掉原条目**（同 id 搬到核心）；降层反之。id 前缀随层改（`p12` → `u12`）时在报告里记映射，面板历史可追。
- 悬而未决：`close` 的删除；`t` 超过 14 天未动的降到当天日记「## 未结」小节。
- 最近要点：新段 `recent` 插在最前，只保留 `caps.RECENT_SEGMENTS` 段，每段末尾标 `（{日期} {段标题}）`。

```python
def render_index(files: dict[str, str], daily: list[str], timeline_lines: int, cap: int = caps.MEM_INDEX) -> tuple[str, str]:
    """返回 (MEMORY.md「记忆目录」节正文, memory/index.md 全文)。files：长期文件相对路径 → 全文。"""
    rows = []
    for rel, text in files.items():
        head = items.header(text)                          # {"title","summary","keywords"}
        its = items.parse_lines(text)
        last = max((i.lh or i.t for i in its), default="")
        rows.append((rel, head, len(its), last))
    people = [r for r in rows if r[0].startswith("memory/people/")]
    topics = [r for r in rows if r[0].startswith("memory/topics/")]
    fmt = lambda r: f"{r[1]['title']}（{r[2]} 条，{r[3][5:] or '—'}）"
    lines = ["（自动生成。用 memory_search 查，或 memory_get 读整份）"]
    if people: lines.append("- 人物：" + " · ".join(fmt(r) for r in sorted(people, key=lambda r: -r[2])))
    if topics: lines.append("- 主题：" + " · ".join(fmt(r) for r in sorted(topics, key=lambda r: -r[2])))
    if daily: lines.append(f"- 日记：{daily[-1][7:12]} 至 {daily[0][7:12]} 共 {len(daily)} 天 · 时间线 {timeline_lines} 行")
    short = "\n".join(lines)
    while len(short) > cap and len(lines) > 2:               # 超预算：主题只留条目最多的前 N 个
        topics = sorted(topics, key=lambda r: -r[2])[:-1]
        lines[2:3] = ["- 主题：" + " · ".join(fmt(r) for r in topics) + " ……其余见 memory/index.md"] if topics else []
        short = "\n".join(lines)
    full = ["# 记忆全目录（自动生成，不要手改）"]
    for rel, head, n, last in sorted(rows):
        full.append(f"## {rel}\n> {head['summary']}\n> 关键词：{'、'.join(head['keywords'])}\n> {n} 条，最近 {last}")
        full += [f"- {s}" for s in items.section_titles(files[rel])]
    return short, "\n".join(full)[:caps.INDEX_FULL]
```

#### render.py（要点）

- `apply_ops(files, ops, today)`：按 `file` 把 op 落到对应文件/节：`core.user.*` → USER.md 对应分节（前置注释格式），`core.facts` → MEMORY「关键事实」，`people/*`、`topics/*` → 长期文件（行尾元数据；文件不存在就用 `headers` 里的摘要/关键词新建）；`add` 分配本文件最大号 +1；`rewrite` 原地换文本、`t` 不变、`src` 换成新证据；`delete` 连元数据一起删。旧的 `status: superseded` 条目第一次整理时一并清掉。
- `apply_headers(files, headers)`：只改长期文件头的摘要与关键词（关键词去重，≤12 个）。
- `apply_insights(files, insights)`：写进 `memory/people/<人>.md` 的 `## 洞察` 小节（不进核心）。
- `append_daily(text, seg)`：每日笔记追加 `## 段 HH:MM–HH:MM · {标题}`（事件/决定/承诺/未结），行尾带 `src`。
- `append_timeline(text, items)`：`memory/timeline.md` 只追加 `YYYY-MM-DD HH:MM | 谁 | 类别 | 内容 | src:s:xxxx#n`；超过 30KB 时把最早的整月搬到 `memory/timeline/YYYY-MM.md`。
- `build_plan(out, cur, today, seg, sid) -> (plan, report)`：`apply_ops` → `merge_similar` → `rebalance`/`promote_candidates` → 悬而未决与最近要点 → `apply_headers`/`apply_insights` → 每日笔记、时间线 → `render_index` 重建目录与全目录；`plan` 只含有变化的文件（相对路径 → 新全文），`report` 记升/降/合并/丢弃。新建的家人人物文件称呼另返回 `new_people`，提交时追加进 `guanjia/family-facts.json` 的 `people`（复用 `memory_page.check_facts`）。
- `files_for(fs, msgs)`：本次要读写的文件 = USER.md、MEMORY.md、`relevant.pick(msgs)` 选中的长期文件（合计 ≤`caps.RELEVANT_INPUT`）、当天日记、timeline、index.md；runner 里的 `WRITABLE` 就是它的返回值（升降可能要写没被选中的长期文件：`long_target` 指到的文件按需追加进 `cur` 并读取 sha）。

```python
@dataclass(frozen=True)
class Seg:
    tokens: int        # 当前 sessionId 全部对话的折算 token
    user_turns: int


@dataclass(frozen=True)
class Decision:
    action: str        # skip / consolidate / reset-only / boot-reset
    reason: str
    reset: bool = False
    forced: bool = False


def decide(now_ms: int, wm: Watermark, seg: Seg | None, j: Journal, s: SleepSettings, *, boot: bool = False) -> Decision:
    """纯函数：只看数字做决定，不调模型、不碰文件。seg 为 None 表示还没空闲到需要算它。"""
    if wm.busy:
        return Decision("skip", "busy")
    idle = (now_ms - wm.last_interaction_ms) / 60000
    forced = s.force_enabled and wm.prompt_tokens >= s.force_tokens and idle >= s.force_idle_min
    if not forced and idle < s.idle_min:
        return Decision("skip", "not-idle")
    if seg is None:
        return Decision("skip", "need-seg")
    if not forced and seg.tokens < s.min_tokens and seg.user_turns < s.min_turns:
        return Decision("skip", "too-small")
    done = j.done_seq(wm.sid, rehearse=s.stage == "rehearse")
    has_new = wm.max_seq > done
    can_reset = s.stage == "reset" and (forced or now_ms - j.last_reset_ms >= s.cooldown_min * 60000)
    if boot and can_reset and has_new:
        return Decision("boot-reset", "boot", True, forced)
    if has_new:
        return Decision("consolidate", "forced" if forced else "idle", can_reset, forced)
    if can_reset:
        return Decision("reset-only", "checkpointed", True, forced)
    return Decision("skip", "cooldown" if s.stage == "reset" else "nothing-new")


class SleepRunner:
    def __init__(self, oc: OpenClaw, fs: WorkspaceFiles, settings_path: Path, journal: Journal, log,
                 consolidator_factory, notify=post_sleep_state, now=lambda: int(time.time() * 1000),
                 digests=read_digests, staging_root: Path = DATA / "staging") -> None:
        ...

    # ---- 入口 ----
    def check(self, boot: bool = False) -> str:
        s = load_settings(self.settings_path)
        if s.stage == "off":
            return "off"
        with run_lock(DATA / "run.lock", timeout=0) as got:
            if not got:
                return "locked"
            self._resume(s)                       # journal 停在 staged → 直接提交；snapshot/consolidating → 丢弃重做
            self._closed_backlog(s)               # 换新后留下的已关闭段先补整理（固定 sid，不怕新输入）
            wm = self.oc.watermark(MAIN)
            d = decide(self.now(), wm, None, self.j, s, boot=boot)
            if d.reason == "need-seg":
                d = decide(self.now(), wm, self._seg(wm), self.j, s, boot=boot)
            self.log.emit("sleep", "sleep.check", d.reason, action=d.action, sid=wm.sid[:8], seq=wm.max_seq,
                          idle_min=round((self.now() - wm.last_interaction_ms) / 60000), stage=s.stage)
            if d.action == "skip":
                return d.reason
            if d.action == "boot-reset":
                return self._boot_reset(wm, s)
            if d.action == "consolidate":
                ok = self._consolidate(wm.sid, self.j.done_seq(wm.sid, rehearse=s.stage == "rehearse"), wm.max_seq, s)
                if not (ok and d.reset):
                    return "consolidated" if ok else "failed"
            return self._try_reset(wm)

    def quick_reset(self) -> str:
        """语音进程启动时后台调用：只做「临时交接 + 换新」，不调模型。条件不满足返回 not-needed。"""
        s = load_settings(self.settings_path)
        if s.stage != "reset":
            return "disabled"
        with run_lock(DATA / "run.lock", timeout=0) as got:
            if not got:
                return "locked"
            wm = self.oc.watermark(MAIN)
            d = decide(self.now(), wm, self._seg(wm), self.j, s, boot=True)
            return self._boot_reset(wm, s) if d.action == "boot-reset" else "not-needed"

    # ---- 阶段 ----
    def _seg(self, wm: Watermark) -> Seg:
        msgs = self.oc.export(wm.sid, 0, wm.max_seq)
        return Seg(estimate_tokens("\n".join(m.text for m in msgs)), sum(1 for m in msgs if m.role == "user"))

    def _consolidate(self, sid: str, after: int, upto: int, s: SleepSettings) -> bool:
        msgs = self.oc.export(sid, after, upto)
        stage_dir = self.staging_root / f"{sid[:8]}-{upto}"
        self.j.begin(sid, after, upto, stage_dir)                         # phase = snapshot
        self.notify("consolidating")
        t0 = time.monotonic()
        try:
            cur = {rel: self.fs.read(rel) for rel in files_for(self.fs, msgs)}   # 分层：核心 + 相关长期文件 + 日记/时间线/目录；文本 + sha
            digests = self.digests(since_ms=self.j.last_digest_ms)
            texts = {k: v["text"] for k, v in cur.items()}
            prompt = build_prompt(texts, msgs, digests, today_str())       # 填 USER/MEMORY/INDEX/RELEVANT/DIGESTS/TRANSCRIPT 与 caps
            cons = self.cons_factory(s.model)
            self.j.phase("consolidating")

            def check(out: dict) -> list[str]:
                out, _ = filter_ops(copy.deepcopy(out), msgs, digests, sid)
                plan, _ = build_plan(out, texts, today_str(), seg_title(msgs), sid)
                return size_problems(plan) + delete_problems(texts, plan)

            out, _ = cons.run(prompt, check)
            out, dropped = filter_ops(out, msgs, digests, sid)
            plan, report = build_plan(out, texts, today_str(), seg_title(msgs), sid)
            for rel in plan.keys() - cur.keys():                            # 升降/新建碰到的、没预先读的长期文件
                cur[rel] = self.fs.read(rel) if self.fs.resolve(rel).exists() else {"text": "", "sha": ""}
            write_staging(stage_dir, prompt_meta(prompt), out, plan, dropped + report)   # 原始 JSON + 渲染后的文件 + 升/降/合并/丢弃
            self.j.phase("staged")
            if s.stage == "rehearse":
                self.j.mark_rehearsed(sid, upto)
                self._log_run("rehearsed", sid, after, upto, out, dropped, t0)
                return False
            commit(self.fs, plan, {k: v["sha"] for k, v in cur.items()})
            self.j.checkpoint(sid, upto)                                   # phase = committed
            self._log_run("committed", sid, after, upto, out, dropped, t0)
            return True
        except (ConsolidateError, Conflict, LockBusy, httpx.HTTPError, ValueError) as e:
            n = self.j.fail(sid, str(e)[:200])
            self.log.emit("sleep", "sleep.fail", str(e)[:200], level="warn", sid=sid[:8], fails=n)
            return False
        finally:
            self.notify("awake")

    def _try_reset(self, w0: Watermark) -> str:
        """S5 复核 + 栅栏 + S6 换新 + 补差。"""
        w1 = self.oc.watermark(MAIN)
        if not w0.same(w1) or w1.busy:
            return "moved"
        self.notify("sleeping")                    # 语音收到后：新的一句先等（≤10 s），见 P4-7
        try:
            time.sleep(0.5)                        # 让正在识别的那一句落定
            w2 = self.oc.watermark(MAIN)
            if not w0.same(w2) or self.oc.active(MAIN):
                return "moved"
            try:
                new_sid = self.oc.reset(MAIN, w0.sid)
            except SessionChanged:
                return "session-changed"
        finally:
            self.notify("awake")
        tail = [m for m in self.oc.export(w0.sid, w0.max_seq, 1 << 62) if m.role in ("user", "assistant")]
        if tail:                                   # S5→S6 之间落进来的话：原文截断挂到交接末尾，登记补整理
            append_tail_to_handover(self.fs, tail[:5])
            self.j.add_closed(w0.sid, w0.max_seq)
        self.j.reset_done(w0.sid, new_sid, self.now())
        append_jsonl(DATA / "resets.jsonl", {"key": MAIN, "old": w0.sid, "new": new_sid, "at": self.now()})
        self.log.emit("sleep", "sleep.reset", "换新", old=w0.sid[:8], new=new_sid[:8], tail=len(tail))
        return "reset"

    def _boot_reset(self, wm: Watermark, s: SleepSettings) -> str:
        """开机先换新：临时交接（最后 8 条原文截断，不调模型）→ reset → 旧段登记为已关闭，后台补整理。"""
        last = [m for m in self.oc.export(wm.sid, max(0, wm.max_seq - 60), wm.max_seq) if m.role in ("user", "assistant")][-8:]
        write_temp_handover(self.fs, last)         # ws_lock 超时 1 s；拿不到锁就跳过临时交接，照样换新
        r = self._try_reset(wm)
        if r == "reset":
            self.j.add_closed(wm.sid, self.j.done_seq(wm.sid, rehearse=False), upto=wm.max_seq)
        return f"boot-{r}"
```

- `commit()`：

```python
def commit(fs: WorkspaceFiles, plan: dict[str, str], base_sha: dict[str, str], actor: str = "sleep") -> None:
    """同一把工作区锁里：先整组比对 sha，任何一个变了整组放弃（模型或面板刚写过，下次重来）；再逐个原子替换。"""
    with ws_lock(timeout=60, holder=actor, path=fs.lock_path):
        for rel in plan:
            if fs.read(rel)["sha"] != base_sha.get(rel, ""):
                raise Conflict(rel)
        for rel, text in plan.items():
            fs.write_unlocked(rel, text, base_sha.get(rel, ""), actor=actor)
```

  多个文件之间不是原子的：中途失败时已写的文件有历史可回滚，journal 停在 `staged`，下次 `_resume` 用**新的** sha 重建计划再提交（不重调模型），保证最终一致。
- `post_sleep_state(state)`：`POST http://127.0.0.1:18795/sleep-state`，0.5 s 超时，失败忽略（语音没开不影响休眠）。
- 连续失败 3 次同一段：日志 `level=error`，面板休眠卡标红；`stage=reset` 时改走「临时交接 + 换新」，避免主线无限增长（v2 §9）。
- 分层版里「交接」就是 MEMORY.md「最近要点」的第一段：`write_temp_handover` 把最后 8 条原文截断（每条 ≤200 字）作为新的第一段、标「（上一段整理中）」，`append_tail_to_handover` 追加到第一段末尾并标「换新前最后几句（未整理）」；该段正式整理提交时整段替换。
- `run_lock`：复用 `wslock` 的 msvcrt 写法，路径 `data/sleep/run.lock`，`timeout=0` 抢不到立即返回。

```ts
import fs from "node:fs";
import path from "node:path";

type Entry = { rel: string; title: string; summary: string; keywords: string[]; items: { id: string; text: string; imp: number }[] };
const LINE = /^- (.+?)\s*<!--\s*id:\s*(\w+)[^>]*?imp:\s*(\d+)[^>]*-->\s*$/;
const MAX_FILES = 2, MAX_ITEMS = 6, MAX_CHARS = 800;

export function createPrefetch(workspaceDir: string, auditPath: string) {
  const roots = ["memory/people", "memory/topics"].map((d) => path.join(workspaceDir, d));
  let cache: { sig: string; entries: Entry[] } | undefined;

  function load(): Entry[] {
    const files = roots.flatMap((r) => (fs.existsSync(r) ? fs.readdirSync(r).filter((f) => f.endsWith(".md")).map((f) => path.join(r, f)) : []));
    const sig = files.map((f) => `${f}:${fs.statSync(f).mtimeMs}`).join("|");
    if (cache?.sig === sig) return cache.entries;
    const entries = files.map((f) => {
      const lines = fs.readFileSync(f, "utf8").split(/\r?\n/);
      const title = (lines.find((l) => l.startsWith("# ")) ?? "").slice(2).trim();
      const summary = (lines.find((l) => l.startsWith("> 摘要：")) ?? "").slice(5).trim();
      const kw = (lines.find((l) => l.startsWith("> 关键词：")) ?? "").slice(6).split(/[、,，\s]+/).filter((k) => k.length >= 2);
      const items = lines.map((l) => LINE.exec(l)).filter(Boolean).map((m) => ({ text: m![1], id: m![2], imp: Number(m![3]) }));
      return { rel: path.relative(workspaceDir, f).replace(/\\/g, "/"), title, summary, keywords: [title, ...kw].filter(Boolean), items };
    });
    cache = { sig, entries };
    return entries;
  }

  /** 原话命中文件头关键词 → 摘要 + imp 最高的几条；返回 appendContext 文本或 undefined */
  return function prefetch(userText: string, opts: { allowOwner: boolean; allowTopics: boolean } = { allowOwner: true, allowTopics: true }): string | undefined {
    const text = userText.trim();
    if (text.length < 4) return;
    const hits = load()
      .filter((e) => (opts.allowOwner || e.title !== "主人") && (opts.allowTopics || e.rel.startsWith("memory/people/")))
      .map((e) => ({ e, score: e.keywords.filter((k) => text.includes(k)).length }))
      .filter((x) => x.score > 0)
      .sort((a, b) => b.score - a.score)
      .slice(0, MAX_FILES);
    if (!hits.length) return;
    const parts: string[] = [];
    const ids: string[] = [];
    for (const { e } of hits) {
      const top = [...e.items].sort((a, b) => b.imp - a.imp).slice(0, MAX_ITEMS);
      ids.push(...top.map((i) => `${e.rel}#${i.id}`));
      parts.push(`[相关记忆·${e.title}] ${e.summary}${top.length ? "：" + top.map((i) => i.text).join("；") : ""}`);
    }
    try { fs.appendFileSync(auditPath, JSON.stringify({ t: Date.now(), ids }) + "\n", "utf8"); } catch {}
    return parts.join("\n").slice(0, MAX_CHARS);
  };
}
```

**验收**：
1. 一天内出现 ≥1 次 `sleep.reset`，冷却生效（两次间隔 ≥2 小时），每天 ≤4 次。
2. 换新后第一句语音：首字比换新前快（前缀短了），且能接上一段的话头（问「刚才说到哪了」能答出交接里的事）。
3. **主人真重启（E17）**：关机或注销 → 登录 → 立刻说「管家，刚才我们聊到哪了」；另一次登录后等 2 分钟再说。判定：`sleep.fence` 等待 ≤3 s（或没有等）、首句落在新 sid、2 分钟内 MEMORY.md 的临时交接被正式交接替换。
4. `sleep.fence` 超时（≥10 s）一周 0 次。
**风险**：`gateway call` 冷启动 node 的 2–3 s 里主人刚好开口——语音侧被栅栏拦住；微信侧拦不住（P3 后），主人微信刚好在这几秒到达会被 reset 打断：概率低（主人已空闲 60 分钟），记事件，后续可把 RPC 换成常驻 WebSocket 客户端。
