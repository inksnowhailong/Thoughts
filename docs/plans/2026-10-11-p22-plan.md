## B1 声纹库 v2 + 面板引导重新注册【线上数据】

### 设计要点（执行者必读）

- **库格式 v2**（仍是 `data\speakers.npz`，切模型时库名规则不变）：键 `version`=`[2]`、`names`=Unicode 字符串数组、`embs`=float32 `[M,192]`（每段一个已归一化向量）、`owner`=int32 `[M]`（属于 `names` 里第几个人）。不用 pickle。
- **旧库兼容**：没有 `version` 键就是旧格式（每个人一个 `(192,)` 向量）→ 读成 `[1,192]`，`legacy=True`。只读不改写；第一次保存时整库写成 v2。
- **打分取 top-k 平均，k=3**：`score = mean(最大的 min(3,N) 个余弦)`。理由：注册样本刻意分三种情况（近处 / 2–3 米外 / 大声带情绪），一句远处的话只会和「远处」那几段像；对全部样本取平均会被另外两组拉低，正好重演「越远越认不出」。取单个最大值又容易被某一段噪声样本放进来。k=3 约等于一组样本，兼顾两头。N=1（旧库）时就是原来的单向量余弦，分数口径不变。
- **注册走日常链路**：守护进程进「注册模式」后，前端线程把 VAD 段（同一 AEC 设置、×3 增益、同一 VAD）直接交给注册会话，**不识别、不判定、不送主线**；正在播放时的段不收。
- **质量检查**（每段）：总长 ≤10 秒；去掉数字零后的有效语音 ≥1.5 秒；零值占比 <50%；峰值 ≥0.03。不合格面板显示原因，让主人再说一句。
- **重新注册 = 替换这个人的全部样本**（旧向量是 AEC 时代的、恰恰是问题所在，不能混进来）。三组 4+4+3=11 段，至少 8 段才能进自测；自测一段**不计入**库，面板显示「新库分数 / 旧库分数」。
- **备份**：保存前，若 `speakers.campp.pre-reenroll.npz` 不存在就复制一份（只建一次，永远保留最初的旧库）；每次保存还覆盖一份 `speakers.prev.npz`。面板「回滚到注册前」= 用 pre-reenroll 原子替换并热重载。
- 顺带修诊断第 6 节：docstring 过时（默认模型、阈值 0.5）、`SpeakerGate` 默认门槛改成取 `Settings().spk_threshold`、`enroll()` 每段写盘改成整批一次写盘、`enroll.py` 的过时提示语。

import shutil
import threading

from guanjia.settings import Settings

TOP_K = 3
STORE_VERSION = 2
DEFAULT_THRESHOLD = Settings().spk_threshold


def backup_path(store: Path) -> Path:
    """重新注册前的旧库备份：只建一次，回滚用。"""
    name = "speakers.campp.pre-reenroll.npz" if store.name == "speakers.npz" else f"{store.stem}.pre-reenroll.npz"
    return store.with_name(name)


def score_against(v: np.ndarray, refs: np.ndarray, k: int = TOP_K) -> float:
    """v 与一个人全部样本的余弦，取最大的 k 个求平均。"""
    sims = refs @ v
    k = min(k, len(sims))
    return float(np.sort(sims)[-k:].mean())


def load_store(path: Path, dim: int) -> tuple[dict[str, np.ndarray], bool]:
    """读库：返回 ({名字: [N, dim]}, 是否旧格式)。维度不符抛 ValueError。"""
    with np.load(path, allow_pickle=False) as z:
        files = set(z.files)
        if "version" in files:
            names = [str(n) for n in z["names"]]
            embs, owner = z["embs"].astype(np.float32), z["owner"]
            db = {n: embs[owner == i] for i, n in enumerate(names)}
            legacy = False
        else:
            db = {k: np.asarray(z[k], dtype=np.float32).reshape(1, -1) for k in files}
            legacy = True
    bad = [k for k, v in db.items() if v.ndim != 2 or v.shape[1] != dim or len(v) == 0]
    if bad:
        raise ValueError(f"声纹库 {path.name} 与模型维度 {dim} 不符：{bad}")
    return db, legacy


def save_store(path: Path, db: dict[str, np.ndarray]) -> None:
    """整库写成 v2：同目录临时文件 + os.replace。"""
    names = list(db)
    dim = next(iter(db.values())).shape[1] if db else 0
    embs = np.concatenate([db[n] for n in names]).astype(np.float32) if names else np.zeros((0, dim), np.float32)
    owner = np.concatenate([np.full(len(db[n]), i, np.int32) for i, n in enumerate(names)]) if names else np.zeros(0, np.int32)
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_name(f"{path.name}.{os.getpid()}.tmp")
    try:
        with open(tmp, "wb") as f:  # 传文件对象，np.savez 才不会自己补 .npz 后缀
            np.savez(f, version=np.array([STORE_VERSION], np.int32), names=np.array(names, dtype=str), embs=embs, owner=owner)
        os.replace(tmp, path)
    finally:
        tmp.unlink(missing_ok=True)


class SpeakerGate:
    def __init__(self, threshold: float = DEFAULT_THRESHOLD, store: Path = STORE, model: Path = SPK_MODEL) -> None:
        cfg = sherpa_onnx.SpeakerEmbeddingExtractorConfig(model=str(model), num_threads=1, provider="cpu")
        self._ex = sherpa_onnx.SpeakerEmbeddingExtractor(cfg)
        self.model = model
        self.threshold = threshold
        self.store = store
        self.legacy = False
        self._db: dict[str, np.ndarray] = {}  # 整体替换，不原地改：前端线程打分时拿到的总是完整的一份
        self._lock = threading.Lock()  # 只串行化写盘/换库
        self._load()

    @property
    def dim(self) -> int:
        return self._ex.dim

    def embed(self, samples: np.ndarray, sr: int = SR) -> np.ndarray:
        ...  # 原样

    def _load(self) -> None:
        if not self.store.exists():
            self._db, self.legacy = {}, False
            return
        try:
            self._db, self.legacy = load_store(self.store, self._ex.dim)
        except (OSError, ValueError) as e:  # 库和模型不配套或文件坏了：按空库处理（守护进程会告警）
            log.warning("声纹库不可用，按空库处理，请重新注册：%s", e)
            self._db, self.legacy = {}, False

    def reload(self) -> None:
        with self._lock:
            self._load()

    def names(self) -> list[str]:
        return list(self._db)

    def counts(self) -> dict[str, int]:
        return {k: len(v) for k, v in self._db.items()}

    def refs(self, name: str) -> np.ndarray | None:
        return self._db.get(name)

    def set_person(self, name: str, embs: np.ndarray) -> None:
        """用 embs（[N, dim]，已归一化）替换这个人的全部样本，整库一次写盘。"""
        with self._lock:
            db = dict(self._db)
            db[name] = np.asarray(embs, dtype=np.float32)
            save_store(self.store, db)
            self._db, self.legacy = db, False

    def backup(self) -> Path | None:
        """保存前调用：首次建 pre-reenroll 备份（永不覆盖），每次刷新 .prev。库不存在返回 None。"""
        if not self.store.exists():
            return None
        b = backup_path(self.store)
        if not b.exists():
            shutil.copy2(self.store, b)
        shutil.copy2(self.store, self.store.with_name(f"{self.store.stem}.prev.npz"))
        return b

    def rollback(self) -> bool:
        """用 pre-reenroll 备份替换当前库并重载。没有备份返回 False。"""
        b = backup_path(self.store)
        if not b.exists():
            return False
        with self._lock:
            tmp = self.store.with_name(f"{self.store.name}.{os.getpid()}.tmp")
            shutil.copy2(b, tmp)
            os.replace(tmp, self.store)
            self._load()
        return True

    def identify(self, samples: np.ndarray, sr: int = SR) -> tuple[str | None, float]:
        db = self._db
        if not db:
            return None, 0.0
        v = self.embed(samples, sr)
        best, score = None, -1.0
        for name, refs in db.items():
            c = score_against(v, refs)
            if c > score:
                best, score = name, c
        return (best if score >= self.threshold else None), score

SESSION_TIMEOUT_SEC = 600.0


def quality(samples: np.ndarray, sr: int = SR) -> tuple[bool, str, dict]:
    """一段能不能拿来注册。数字零是音箱 DSP 噪声门抹掉的部分，不算有效语音。"""
    n = len(samples)
    nz = int(np.count_nonzero(samples))
    info = {"sec": round(n / sr, 2), "effective_sec": round(nz / sr, 2),
            "zero_ratio": round(1 - nz / n, 2) if n else 1.0, "peak": round(float(np.abs(samples).max()), 3) if n else 0.0}
    if n / sr > MAX_SEG_SEC:
        return False, "这段太长了（可能混进了别的声音），说一句 3～5 秒的话", info
    if nz / sr < MIN_EFFECTIVE_SEC:
        return False, f"有效语音只有 {nz / sr:.1f} 秒，请说长一点（3～5 秒）", info
    if info["zero_ratio"] >= MAX_ZERO_RATIO:
        return False, "声音太小，被音箱截掉了一大半，大声一点或近一点", info
    if info["peak"] < MIN_PEAK:
        return False, "声音太小", info
    return True, "", info


@dataclass
class Sample:
    group: str
    emb: np.ndarray
    info: dict


class EnrollSession:
    def __init__(self, name: str, embed: Callable[[np.ndarray], np.ndarray], *, old_refs: np.ndarray | None,
                 now: Callable[[], float] = time.monotonic) -> None:
        self.name = name
        self._embed = embed
        self._old = old_refs
        self._now = now
        self.started = now()
        self.state = "recording"
        self.samples: list[Sample] = []
        self.last: dict | None = None
        self.test_score: float | None = None
        self.test_old_score: float | None = None
        self._lock = threading.Lock()  # feed 在前端线程，其余在控制口线程

    def _group(self) -> tuple | None:
        for g in GROUPS:
            if sum(1 for s in self.samples if s.group == g[0]) < g[3]:
                return g
        return None

    def expired(self) -> bool:
        return self._now() - self.started > SESSION_TIMEOUT_SEC

    def feed(self, samples: np.ndarray, *, playing: bool = False) -> dict:
        with self._lock:
            if self.state not in ("recording", "testing"):
                return {"ok": False, "reason": "not_active"}
            if playing:
                self.last = {"ok": False, "reason": "电脑正在出声，这段不算"}
                return self.last
            ok, reason, info = quality(samples)
            if not ok:
                self.last = {"ok": False, "reason": reason, "info": info}
                return self.last
            v = self._embed(samples)
            if self.state == "recording":
                g = self._group()
                self.samples.append(Sample(g[0], v, info))
                if self._group() is None:
                    self.state = "testing"
                self.last = {"ok": True, "group": g[0], "info": info}
            else:
                self.test_score = round(score_against(v, self.embs()), 3)
                if self._old is not None:
                    self.test_old_score = round(score_against(v, self._old), 3)
                self.state = "ready"
                self.last = {"ok": True, "group": "test", "info": info}
            return self.last

    def discard_last(self) -> bool:
        with self._lock:
            if self.state != "recording" or not self.samples:
                return False
            self.samples.pop()
            return True

    def to_testing(self) -> bool:
        """「这些够了」：至少 MIN_TOTAL 段才能去自测。"""
        with self._lock:
            if self.state != "recording" or len(self.samples) < MIN_TOTAL:
                return False
            self.state = "testing"
            return True

    def embs(self) -> np.ndarray:
        return np.stack([s.emb for s in self.samples])

    def snapshot(self) -> dict:
        with self._lock:
            g = self._group() if self.state == "recording" else None
            return {"name": self.name, "state": self.state, "total": len(self.samples), "min_total": MIN_TOTAL,
                    "group": g[0] if g else None, "prompt": g[1] if g else None, "examples": list(g[2]) if g else [],
                    "groups": [{"key": k, "prompt": p, "have": sum(1 for s in self.samples if s.group == k), "need": n}
                               for k, p, _, n in GROUPS],
                    "last": self.last, "test_score": self.test_score, "test_old_score": self.test_old_score,
                    "left_sec": max(0, round(SESSION_TIMEOUT_SEC - (self._now() - self.started)))}
```

    def _ensure_gate(self) -> SpeakerGate:
        """注册用的声纹门：平时那把；库为空时守护进程没建门，就另建一把，保存后接上。"""

    def _speaker_info(self, query: dict) -> tuple[int, dict]:
        g = self.gate or self._spare_gate
        return 200, {"ok": True, "model": SPK_MODEL.stem, "store": STORE.name, "people": g.counts() if g else {},
                     "legacy": bool(g and g.legacy), "backup": backup_path(STORE).exists(),
                     "threshold": self.store.current.spk_threshold,
                     "session": self.enroll.snapshot() if self.enroll else None}

    def _speaker_action(self, body: dict) -> tuple[int, dict]:
        """面板「声纹」卡片：start / discard_last / enough / cancel / commit / rollback。"""
        action = str(body.get("action") or "")
        sess = self.enroll
        if action == "start":
            if not self.speaker_gate:
                return 400, {"ok": False, "reason": "声纹已关闭（--no-gate），不能注册"}
            if sess is not None:
                return 409, {"ok": False, "reason": "已经在注册了"}
            if self.state == "SPEAKING" or not self.text_q.empty():
                return 409, {"ok": False, "reason": "正在回话，等它说完再开始"}
            name = str(body.get("name") or "我").strip()[:8] or "我"
            gate = self._ensure_gate()
            self.enroll = EnrollSession(name, gate.embed, old_refs=gate.refs(name))
            log.info("enroll start: %s", name)
            return 200, {"ok": True, "session": self.enroll.snapshot()}
        if action == "rollback":
            if sess is not None:
                return 409, {"ok": False, "reason": "先结束当前注册"}
            gate = self._ensure_gate()
            if not gate.rollback():
                return 404, {"ok": False, "reason": "没有注册前的备份"}
            if self.gate is None and gate.names():
                self.gate = gate
            log.warning("speaker store rolled back to %s", backup_path(STORE).name)
            return 200, {"ok": True, "people": gate.counts(), "legacy": gate.legacy}
        if sess is None:
            return 409, {"ok": False, "reason": "没有在注册"}
        if action == "cancel":
            self.enroll = None
            log.info("enroll cancelled")
            return 200, {"ok": True}
        if action == "discard_last":
            sess.discard_last()
        elif action == "enough":
            if not sess.to_testing():
                return 409, {"ok": False, "reason": "至少要 8 段才能自测"}
        elif action == "commit":
            if sess.state != "ready":
                return 409, {"ok": False, "reason": "先说一句自测"}
            gate = self._ensure_gate()
            backup = gate.backup()
            gate.set_person(sess.name, sess.embs())
            if self.gate is None:
                self.gate = gate
            self.enroll = None
            log.info("enroll saved: %s n=%d test=%.3f old=%s backup=%s", sess.name, len(sess.samples), sess.test_score,
                     sess.test_old_score, backup.name if backup else "-")
            return 200, {"ok": True, "people": gate.counts(), "test_score": sess.test_score,
                         "test_old_score": sess.test_old_score}
        else:
            return 400, {"ok": False, "reason": f"不认识的操作：{action}"}
        return 200, {"ok": True, "session": sess.snapshot()}
   ```

### `src/guanjia/wslock.py`（完整）

```python
"""工作区写锁：面板编辑与 P4 休眠提交共用。锁文件锁第 0 字节（msvcrt.locking），进程退出系统自动释放；不可重入。

    with ws_lock(holder="sleep"):          # P4：一次提交多个文件
        fs.write_unlocked("USER.md", ...)
        fs.write_unlocked("MEMORY.md", ...)
"""

from __future__ import annotations

import json
import msvcrt
import os
import time
from collections.abc import Iterator
from contextlib import contextmanager
from pathlib import Path

LOCK_PATH = Path.home() / ".openclaw" / "workspace" / "guanjia" / "ws-write.lock"


class LockBusy(Exception):
    """等超时还拿不到锁；str() 是给人看的持有者信息。"""


def _info_path(path: Path) -> Path:
    return path.with_name(path.name + ".json")


def _holder(path: Path) -> str:
    try:
        d = json.loads(_info_path(path).read_text(encoding="utf-8"))
        return f"{d.get('holder')}（pid {d.get('pid')}，{d.get('since')} 起）"
    except (OSError, ValueError):
        return "未知"


@contextmanager
def ws_lock(timeout: float = 10.0, holder: str = "panel", path: Path = LOCK_PATH) -> Iterator[None]:
    path.parent.mkdir(parents=True, exist_ok=True)
    fd = os.open(path, os.O_RDWR | os.O_CREAT, 0o644)
    try:
        deadline = time.monotonic() + timeout
        while True:
            try:
                os.lseek(fd, 0, os.SEEK_SET)
                msvcrt.locking(fd, msvcrt.LK_NBLCK, 1)
                break
            except OSError:
                if time.monotonic() >= deadline:
                    raise LockBusy(_holder(path)) from None
                time.sleep(0.1)
        try:
            _info_path(path).write_text(json.dumps({"holder": holder, "pid": os.getpid(),
                                                    "since": time.strftime("%H:%M:%S")}), encoding="utf-8")
            yield
        finally:
            os.lseek(fd, 0, os.SEEK_SET)
            msvcrt.locking(fd, msvcrt.LK_UNLCK, 1)
    finally:
        os.close(fd)
```

### `src/guanjia/wsfiles.py`（完整）

```python
"""OpenClaw 工作区文本文件的安全读写（面板提示词页、记忆页；P4 休眠提交复用）。

    读：返回规范成 LF 的文本 + sha（编辑器用）
    写：拿写锁 → 重读比对 base_sha（模型可能刚写过）→ 校验 → 备份旧版 → 原子替换（保持原换行风格）→ 审计
"""

from __future__ import annotations

import hashlib
import json
import re
import time
from collections.abc import Callable
from pathlib import Path

from guanjia.fsutil import atomic_write_text
from guanjia.wslock import LOCK_PATH, ws_lock

WORKSPACE = Path.home() / ".openclaw" / "workspace"
HISTORY_DIR = Path(__file__).resolve().parents[2] / "data" / "ws-history"
HARD_MAX_CHARS = 20000
HISTORY_KEEP = 20
ALLOWED = ("SOUL.md", "IDENTITY.md", "AGENTS.md", "USER.md", "MEMORY.md", "memory/timeline.md",
           "skills/guanjia-notes/SKILL.md", "skills/guanjia-lookup/SKILL.md", "skills/guanjia-launch/SKILL.md",
           "guanjia/family-facts.json")
_DAILY = re.compile(r"^memory/\d{4}-\d{2}-\d{2}\.md$")


class Conflict(Exception):
    """文件在打开之后被改过（多半是模型刚写过）。"""


def sha(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()[:16]


class WorkspaceFiles:
    def __init__(self, root: Path = WORKSPACE, history_dir: Path = HISTORY_DIR, lock_path: Path = LOCK_PATH) -> None:
        self.root = root
        self.history_dir = history_dir
        self.lock_path = lock_path

    def resolve(self, rel: str) -> Path:
        rel = rel.replace("\\", "/")
        if rel not in ALLOWED and not _DAILY.match(rel):
            raise ValueError(f"不允许的文件：{rel}")
        p = (self.root / rel).resolve()
        if self.root.resolve() not in p.parents:
            raise ValueError(f"不允许的文件：{rel}")
        return p

    def _raw(self, p: Path) -> str | None:
        return p.read_bytes().decode("utf-8-sig") if p.exists() else None

    def read(self, rel: str) -> dict:
        p = self.resolve(rel)
        raw = self._raw(p)
        text = (raw or "").replace("\r\n", "\n")
        return {"rel": rel, "exists": raw is not None, "text": text, "sha": sha(raw) if raw is not None else "",
                "chars": len(text), "mtime": p.stat().st_mtime if raw is not None else None}

    def write(self, rel: str, text: str, base_sha: str, *, actor: str = "panel",
              validate: Callable[[str | None, str], None] | None = None) -> dict:
        with ws_lock(holder=actor, path=self.lock_path):
            return self.write_unlocked(rel, text, base_sha, actor=actor, validate=validate)

    def write_unlocked(self, rel: str, text: str, base_sha: str, *, actor: str = "panel",
                       validate: Callable[[str | None, str], None] | None = None) -> dict:
        """调用方必须已持有 ws_lock。validate(旧文本或 None, 新文本) 不通过就抛 ValueError。"""
        p = self.resolve(rel)
        text = text.replace("\r\n", "\n")
        if len(text) > HARD_MAX_CHARS:
            raise ValueError(f"太长了：{len(text)} 字，上限 {HARD_MAX_CHARS}")
        raw = self._raw(p)
        if (sha(raw) if raw is not None else "") != base_sha:
            raise Conflict(rel)
        old = raw.replace("\r\n", "\n") if raw is not None else None
        if validate:
            validate(old, text)
        if old == text:
            return self.read(rel)
        if raw is not None:
            self._backup(rel, raw)
        out = text.replace("\n", "\r\n") if raw is not None and "\r\n" in raw else text
        p.parent.mkdir(parents=True, exist_ok=True)
        atomic_write_text(p, out)
        self._audit(rel, actor, sha(raw) if raw is not None else "", sha(out), len(text))
        return self.read(rel)

    def _hist_dir(self, rel: str) -> Path:
        return self.history_dir / rel.replace("/", "__")

    def _backup(self, rel: str, raw: str) -> None:
        d = self._hist_dir(rel)
        d.mkdir(parents=True, exist_ok=True)
        (d / f"{time.strftime('%Y%m%d-%H%M%S')}-{time.time_ns() % 10**9:09d}.txt").write_text(raw, encoding="utf-8", newline="")
        for f in sorted(d.glob("*.txt"))[:-HISTORY_KEEP]:
            f.unlink(missing_ok=True)

    def _audit(self, rel: str, actor: str, old: str, new: str, chars: int) -> None:
        self.history_dir.mkdir(parents=True, exist_ok=True)
        line = {"t": time.strftime("%Y-%m-%dT%H:%M:%S%z"), "file": rel, "actor": actor, "old": old, "new": new, "chars": chars}
        with open(self.history_dir / "audit.jsonl", "a", encoding="utf-8") as f:
            f.write(json.dumps(line, ensure_ascii=False) + "\n")

    def history(self, rel: str) -> list[dict]:
        self.resolve(rel)
        d = self._hist_dir(rel)
        files = sorted(d.glob("*.txt"), reverse=True) if d.exists() else []
        return [{"id": f.stem, "t": f.stem[:15], "bytes": f.stat().st_size} for f in files]

    def history_text(self, rel: str, version: str) -> str:
        self.resolve(rel)
        if not re.fullmatch(r"\d{8}-\d{6}-\d{9}", version):
            raise ValueError("版本号不对")
        return (self._hist_dir(rel) / f"{version}.txt").read_text(encoding="utf-8").replace("\r\n", "\n")

    def daily_notes(self, limit: int = 30) -> list[str]:
        d = self.root / "memory"
        names = sorted((f.name for f in d.glob("*.md") if _DAILY.match(f"memory/{f.name}")), reverse=True) if d.exists() else []
        return [f"memory/{n}" for n in names[:limit]]
```

readonly（只展示）/ setting（在参数里改）。"""

from __future__ import annotations

import logging
import re
from dataclasses import asdict, dataclass
from datetime import datetime
from pathlib import Path

log = logging.getLogger("guanjia.prompts")
REPO = Path(__file__).resolve().parents[2]
DEFAULT_DIR = Path(__file__).with_name("prompts")

def render(template: str, ctx: dict[str, str]) -> tuple[str, list[str]]:
    """替换 {{占位符}}；返回 (结果, 未知占位符)。未知的原样留着。"""
    unknown: list[str] = []

    def sub(m: re.Match) -> str:
        if m.group(1) in ctx:
            return ctx[m.group(1)]
        unknown.append(m.group(1))
        return m.group(0)

    return _PH.sub(sub, template), unknown


def voice_ctx(now: datetime) -> dict[str, str]:
    return {"DATE": f"{now:%Y-%m-%d}", "TIME": f"{now:%H:%M}", "WEEKDAY": _WEEKDAY_CN[now.weekday()]}


def load_template(pid: str) -> tuple[str, bool]:
    """(模板文本, 是否被覆盖)。覆盖版在 data/prompts/<id>.txt。"""
    o = OVERRIDE_DIR / f"{pid}.txt"
    if o.exists():
        return o.read_text(encoding="utf-8-sig").strip("\r\n"), True
    return (DEFAULT_DIR / f"{pid}.txt").read_text(encoding="utf-8").strip("\r\n"), False


def check_template(pid: str, text: str) -> str:
    """校验并返回渲染预览；不合格抛 ValueError。"""
    rendered, unknown = render(text, voice_ctx(datetime(2026, 10, 11, 18, 5)) if pid == "voice_tag" else {})
    if unknown:
        raise ValueError(f"不认识的占位符：{', '.join(unknown)}（可用：{', '.join(PLACEHOLDERS.get(pid, {}))}）")
    if pid == "voice_tag" and not VOICE_TAG_RULE.match(rendered):
        raise ValueError("语音标记必须是一行、以「[语音 」开头、以「]」结尾、括号里不超过 40 字")
    return rendered


def save_template(pid: str, text: str) -> None:
    from guanjia.fsutil import atomic_write_text

    check_template(pid, text)
    OVERRIDE_DIR.mkdir(parents=True, exist_ok=True)
    atomic_write_text(OVERRIDE_DIR / f"{pid}.txt", text.strip("\r\n"))


def reset_template(pid: str) -> None:
    (OVERRIDE_DIR / f"{pid}.txt").unlink(missing_ok=True)


def render_voice_tag(now: datetime | None = None) -> str:
    """brain.voice_tag 用。覆盖文件被手工改坏时回退内置格式（保护插件和语音规则）。"""
    now = now or datetime.now()
    try:
        text, _ = load_template("voice_tag")
        out, unknown = render(text, voice_ctx(now))
        if not unknown and VOICE_TAG_RULE.match(out):
            return out
        log.warning("voice_tag 模板不合格，用内置格式：%r", text)
    except OSError as e:
        log.warning("voice_tag 模板读不了，用内置格式：%s", e)
    return f"[语音 {now:%Y-%m-%d %H:%M} 周{_WEEKDAY_CN[now.weekday()]}]"


def split_block(text: str) -> tuple[str, str, str]:
    """AGENTS.md → (段前含开始标记, 段内, 结束标记含段后)。标记缺失或重复抛 ValueError。"""
    if text.count(BLOCK_BEGIN) != 1 or text.count(BLOCK_END) != 1 or text.index(BLOCK_BEGIN) > text.index(BLOCK_END):
        raise ValueError("AGENTS.md 里管家规则的开始/结束标记不完整，先人工修好")
    a = text.index(BLOCK_BEGIN) + len(BLOCK_BEGIN)
    b = text.index(BLOCK_END)
    return text[:a], text[a:b], text[b:]


def entries() -> list[dict]:
    return [asdict(e) for e in PROMPTS]

### `params.py`（描述即可）
