```ts
import fs from "node:fs";
import path from "node:path";

export type Person = { id: string; name: string; role: "owner" | "family"; channel: string; accountId: string; userId: string };
export type Source = { kind: "voice" | "owner-wechat" | "family" | "unknown"; person?: Person };

const MAIN = "agent:main:main";

/** people.json 每次按修改时间重读（改完即时生效，同 levels.json） */
export function createPeople(workspaceDir: string, defaultsDir: string) {
  const live = path.join(workspaceDir, "guanjia", "people.json");
  let cache: { mtime: number; list: Person[] } | undefined;
  function list(): Person[] {
    const file = fs.existsSync(live) ? live : path.join(defaultsDir, "people.json");
    const mtime = fs.statSync(file).mtimeMs;
    if (!cache || cache.mtime !== mtime) {
      const raw = JSON.parse(fs.readFileSync(file, "utf8"));
      cache = { mtime, list: (raw.people ?? []).filter((p: Person) => p && p.id && p.accountId && p.userId) };
    }
    return cache.list;
  }
  return {
    list,
    byAccount(accountId?: string, from?: string): Person | undefined {
      const same = list().filter((p) => p.accountId === accountId);
      if (!from) return same.length === 1 ? same[0] : undefined;
      return same.find((p) => p.userId === from) ?? same.find((p) => p.userId.toLowerCase() === from.toLowerCase());
    },
    byName(name: string): Person | undefined {
      const n = name.trim();
      return list().find((p) => p.name === n || p.id === n || (p as any).aliases?.includes(n) || (n === "我" && p.role === "owner"));
    },
    /** 认人：语音 HTTP 轮 requester.channel 是 webchat；微信轮按 accountId 查表 */
    source(ctx: any): Source {
      const ch = ctx?.requester?.channel;
      if (ch === "webchat" || (!ch && ctx?.sessionKey === MAIN)) return { kind: "voice" };
      const p = this.byAccount(ctx?.requester?.accountId);
      if (!p) return { kind: "unknown" };
      return { kind: p.role === "owner" ? "owner-wechat" : "family", person: p };
    },
  };
}
```

### P3-3 家人摘要静默递进主线（★ 插件 + AGENTS，1–1.5 天）

**新文件** `openclaw/plugins/guanjia-guard/src/digests.ts`

```ts
import fs from "node:fs";
import os from "node:os";
import path from "node:path";

/** 家人摘要日志：仓库外、工作区外（memory_search 扫不到），一天一个 jsonl，只追加事件。P4 休眠把它当整理输入。 */
export const STATE_DIR = path.join(os.homedir(), ".openclaw", "guanjia-state");
const DIGEST_DIR = path.join(STATE_DIR, "digests");
const RESETS = process.env.GUANJIA_RESETS ?? "F:\\coding\\guanjia\\data\\sleep\\resets.jsonl";   // P4 写，插件只读

const MERGE_GRACE_MS = 120_000;   // 比 90 s 合并窗口稍长：还在合并缓冲里的不算失败

export type Digest = { id: string; at: number; person: string; name: string; text: string; kind: "msg" | "tell" };
type Ev = ({ ev: "add" } & Digest) | { ev: "enqueued" | "failed"; id: string; at: number } | { ev: "consumed"; ids: string[]; at: number };

function day(ts: number) { return new Date(ts + 8 * 3600e3).toISOString().slice(0, 10); }
function append(e: Ev) {
  fs.mkdirSync(DIGEST_DIR, { recursive: true });
  fs.appendFileSync(path.join(DIGEST_DIR, `${day(e.at)}.jsonl`), JSON.stringify(e) + "\n", "utf8");
}

export function createDigests(now = () => Date.now()) {
  /** 最近 3 天的状态（摘要量很小，每次现读即可） */
  function state() {
    const items = new Map<string, Digest & { enqueuedAt?: number; consumedAt?: number }>();
    for (let i = 2; i >= 0; i--) {
      const f = path.join(DIGEST_DIR, `${day(now() - i * 86400e3)}.jsonl`);
      if (!fs.existsSync(f)) continue;
      for (const line of fs.readFileSync(f, "utf8").split("\n")) {
        if (!line.trim()) continue;
        let e: Ev; try { e = JSON.parse(line); } catch { continue; }
        if (e.ev === "add") items.set(e.id, { ...e });
        else if (e.ev === "enqueued") { const d = items.get(e.id); if (d) d.enqueuedAt = e.at; }
        else if (e.ev === "consumed") for (const id of e.ids) { const d = items.get(id); if (d) d.consumedAt = e.at; }
      }
    }
    return [...items.values()];
  }
  function lastResetAt(key: string): number {
    if (!fs.existsSync(RESETS)) return 0;
    const lines = fs.readFileSync(RESETS, "utf8").trim().split("\n").reverse();
    for (const l of lines) { try { const r = JSON.parse(l); if (r.key === key) return Number(r.at) || 0; } catch {} }
    return 0;
  }
  return {
    add(d: Omit<Digest, "at">) { const x = { ...d, at: now() }; append({ ev: "add", ...x }); return x; },
    mark(id: string, ev: "enqueued" | "failed") { append({ ev, id, at: now() }); },
    consumed(ids: string[]) { if (ids.length) append({ ev: "consumed", ids, at: now() }); },
    pending() { return state().filter((d) => d.enqueuedAt && !d.consumedAt); },
    /** 入队失败（满 32 条等）又没被消费的：主线下一轮直接带上 */
    failedUnconsumed() { return state().filter((d) => !d.enqueuedAt && !d.consumedAt && now() - d.at > MERGE_GRACE_MS); },
    /** 主线换新会清掉没消费的注入：入队早于最近一次换新、又没被消费的，要在新会话第一轮补上 */
    lostByReset(key: string) { const r = lastResetAt(key); return r ? state().filter((d) => d.enqueuedAt && d.enqueuedAt < r && !d.consumedAt) : []; },
  };
}
```

```ts
const MAIN = "agent:main:main";
const MERGE_MS = 90_000;      // 同一家人 90 秒内的连发合并成一条注入
const LINE_MAX = 120;

export function registerWechat(api: any, people: ReturnType<typeof createPeople>, digests: ReturnType<typeof createDigests>) {
  const buf = new Map<string, { person: Person; lines: string[]; ids: string[]; first: number; timer: NodeJS.Timeout }>();

  async function inject(text: string, key: string, ids: string[]) {
    try {
      await api.session.workflow.enqueueNextTurnInjection({
        sessionKey: MAIN, agentId: "main", text, idempotencyKey: key,
        placement: "prepend",                 // 取值以 docs/plugins/hooks/prompt-and-session.md L306–345 与类型定义为准
        ttlMs: 3 * 86400e3,
      });
      ids.forEach((id) => digests.mark(id, "enqueued"));
    } catch (e: any) {                        // 满 32 条会直接拒绝：留在日志里，主线下一轮由 before_prompt_build 兜底带上
      ids.forEach((id) => digests.mark(id, "failed"));
      api.logger?.warn?.(`[guanjia] 家人摘要入队失败：${e?.message ?? e}`);
    }
  }

  function flush(pid: string) {
    const b = buf.get(pid); if (!b) return;
    buf.delete(pid);
    const span = `${hhmm(b.first)}${b.lines.length > 1 ? "–" + hhmm(Date.now()) : ""}`;
    const text = `[家人消息·${b.person.name}·${span}·${b.lines.length} 条] ${b.lines.join("；").slice(0, 400)}`;
    void inject(text, `wx:${b.ids[0]}`, b.ids);
  }

  // 1) 家人入站：只读观测里做确定性摘要（不调模型）
  api.on("message_received", (event: any, ctx: any) => {
    if ((event?.channelId ?? ctx?.channelId) !== "openclaw-weixin") return;
    const p = people.byAccount(event?.accountId ?? ctx?.accountId, event?.from);
    if (!p || p.role !== "family") return;
    const line = oneLine(String(event?.content ?? "")).slice(0, LINE_MAX);
    if (!line || line.startsWith("/")) return;             // 斜杠命令不转
    const d = digests.add({ id: String(event?.messageId ?? `${p.id}:${Date.now()}`), person: p.id, name: p.name, text: line, kind: "msg" });
    const b = buf.get(p.id);
    if (b) { b.lines.push(line); b.ids.push(d.id); return; }
    buf.set(p.id, { person: p, lines: [line], ids: [d.id], first: d.at, timer: setTimeout(() => flush(p.id), MERGE_MS) });
  });

  // 2) 主线每轮构建提示：标记已消费；换新/入队失败丢掉的，补进本轮
  api.on("before_prompt_build", (_event: any, ctx: any) => {
    if (ctx?.sessionKey !== MAIN) return;
    const lost = [...digests.lostByReset(MAIN), ...digests.failedUnconsumed()];
    digests.consumed(digests.pending().map((d) => d.id).concat(lost.map((d) => d.id)));
    if (!lost.length) return;
    return { prependContext: lost.map((d) => `[家人消息·${d.name}·${hhmm(d.at)}] ${d.text}`).join("\n").slice(0, 1200) };
  }, { priority: 50 });

  // 3) 家人支线：信封（谁在说话）
  api.on("before_prompt_build", (_event: any, ctx: any) => {
    const src = people.source(ctx);
    if (src.kind !== "family") return;
    return { prependContext: `（这条来自家人「${src.person!.name}」的微信。家人与主人同权；不主动提主人分节和主线交接里的事。需要主人知道或决定的，调用 tell_owner。）` };
  }, { priority: 40 });

  // 4) tell_owner：家人支线里把要紧事递给主线（+ 紧急时微信通知主人）
  api.registerTool(createTellOwnerTool({ people, digests, inject, send: sendWeixin }));
}
```

```ts
import { execFile } from "node:child_process";
import path from "node:path";

const NODE = process.execPath;                                   // 网关自己的 node（fnm 固定安装路径下的 node.exe）

/** 唯一的发送实现：渠道、账号、原样目标全部显式（不依赖会话的最后路由）。 */
export async function sendWeixin(api: any, p: Person, text: string): Promise<SendResult> {
  const outbound = api.runtime?.channel?.outbound;
  if (outbound?.send) {                                          // 核实第 4 节：外部插件能否拿到需实测，拿不到走 CLI
    const r = await outbound.send({ channel: p.channel, accountId: p.accountId, to: p.userId, text });
    return { ok: !r?.error, detail: String(r?.error ?? "sent") };
  }
  return await new Promise((resolve) => {
    execFile(NODE, [CLI, "message", "send", "--channel", p.channel, "--account", p.accountId, "--target", p.userId,
                    "--message", text, "--json"], { timeout: 30_000, windowsHide: true },
      (err, stdout, stderr) => resolve({ ok: !err, detail: (err ? String(stderr || err.message) : "sent").slice(0, 200) }));
  });
}

export function createWechatSendTool(deps: { api: any; people: any; outbox: Outbox; lastInbound: (p: Person) => number; audit: (o: object) => void }) {
  return {
    name: "wechat_send",
    description: "给主人或家人发一条微信。to 填「主人」或家人称呼；text 是要发的话（≤500 字）。",
    parameters: { type: "object", properties: { to: { type: "string" }, text: { type: "string" } }, required: ["to", "text"] },
    async execute(_id: string, args: { to: string; text: string }, ctx: any) {
      const p = deps.people.byName(args.to);
      if (!p) return `没有这个人：${args.to}。可选：${deps.people.list().map((x: Person) => x.name).join("、")}`;
      // 主人 10-13 定：给家人发只在主人明确让发时（主人语音/主人微信这一轮）；家人会话里只能发给他自己
      const src = deps.people.source(ctx);
      const allowed = p.role === "owner" || src.kind === "voice" || src.kind === "owner-wechat" ||
                      (src.kind === "family" && src.person?.id === p.id);
      if (!allowed) {
        deps.audit({ tool: "wechat_send", person: p.id, from: ctx?.sessionKey, ok: false, detail: "refused-family-policy" });
        return "给家人发微信要主人明确让发；家人自己设的提醒请在建提醒时直接指定发给他自己";
      }
      const text = String(args.text ?? "").trim().slice(0, 500);
      if (!text) return "没有要发的内容";
      const fresh = Date.now() - deps.lastInbound(p) < WINDOW_MS;
      let r: SendResult;
      if (!fresh) { deps.outbox.put(p, text); r = { ok: true, queued: true, detail: "对方很久没说话，先存着，等他下次发消息时补发" }; }
      else r = await sendWeixin(deps.api, p, text);
      deps.audit({ tool: "wechat_send", channel: p.channel, account: p.accountId, target: mask(p.userId), person: p.id,
                   from: ctx?.sessionKey, ok: r.ok, queued: !!r.queued, detail: r.detail, chars: text.length });
      return r.queued ? `已存进待发箱：${r.detail}` : r.ok ? `已发给${p.name}` : `没发出去：${r.detail}`;
    },
  };
}
```
